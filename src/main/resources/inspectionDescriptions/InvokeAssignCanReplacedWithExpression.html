Reports when INVOKE_ASSIGN could be replaced with a MixinExtras expression.

Expressions are preferred over INVOKE_ASSIGN because they fail when an assignment doesn't exist. 


// bad
@Inject(method = "foo", at = @At(value = "INVOKE_ASSIGN", target = "Lcom/example/Foo;getName()Ljava/lang/String;"))

// good
@Definition(id = "getName", method = "Lcom/example/Foo;getName()Ljava/lang/String;")
@Expression("? = ?.getName()")
@Inject(method = "foo", at = @At(value = "MIXINEXTRAS:EXPRESSION", shift = At.Shift.AFTER))
